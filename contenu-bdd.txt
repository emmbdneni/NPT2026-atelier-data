
-- Suppression des tables de programiz

-- Suppression des tables enquête 
DROP TABLE IF EXISTS crime_scene_report; 
DROP TABLE IF EXISTS solution; 
DROP TABLE IF EXISTS get_fit_now_check_in; 
DROP TABLE IF EXISTS get_fit_now_member; 
DROP TABLE IF EXISTS interview; 
DROP TABLE IF EXISTS person;

-- Création des tables 
CREATE TABLE crime_scene_report (  id INTEGER PRIMARY KEY,  date INTEGER,  type TEXT,  description TEXT,  city TEXT); CREATE TABLE person ( id INTEGER PRIMARY KEY,  name TEXT,  address_number INTEGER,  address_street TEXT ); CREATE TABLE interview (  person_id INTEGER PRIMARY KEY,  transcript TEXT,  FOREIGN KEY (person_id) REFERENCES person(id) ); CREATE TABLE get_fit_now_member (  id TEXT PRIMARY KEY,  person_id INTEGER,  name TEXT,  membership_start_date INTEGER,  membership_status TEXT,  FOREIGN KEY (person_id) REFERENCES person(id)); CREATE TABLE get_fit_now_check_in (  membership_id TEXT,  check_in_date INTEGER,  check_in_time INTEGER,  check_out_time INTEGER,  FOREIGN KEY (membership_id) REFERENCES get_fit_now_member(id) ); CREATE TABLE solution (  user INTEGER,  value TEXT );

-- Création du trigger pour afficher la bonne réponse

ELSE 'Erreur sur le coupable, essayez encore !'                             END); END;
